cmake_minimum_required(VERSION 3.20)
project(MushroomRecompiled C)
if(WIN32)
    enable_language(RC)
endif()
include(CTest)
option(SMB3_PLATFORM_TESTS_ONLY "Build tests that do not need the private ROM" OFF)
option(SMB3_CONTROLLER_TESTS "Test the shared SDL controller code with a virtual pad" OFF)
if(SMB3_CONTROLLER_TESTS)
    find_package(SDL2 REQUIRED)
    add_executable(smb3-controller-verify tests/controller_verify.c src/input.c)
    target_include_directories(smb3-controller-verify PRIVATE src)
    target_link_libraries(smb3-controller-verify PRIVATE SDL2::SDL2)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(smb3-controller-verify PRIVATE -UNDEBUG)
    endif()
    add_test(NAME sdl-controller COMMAND smb3-controller-verify)
endif()
add_executable(smb3-rom-verify tests/rom_validation_verify.c runtime/rom_validation.c)
target_include_directories(smb3-rom-verify PRIVATE runtime)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(smb3-rom-verify PRIVATE -UNDEBUG)
endif()
add_test(NAME rom-validation COMMAND smb3-rom-verify)
add_executable(smb3-android-video-verify tests/android_video_verify.c android/native/video_layout.c)
target_include_directories(smb3-android-video-verify PRIVATE android/native)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(smb3-android-video-verify PRIVATE -UNDEBUG)
endif()
add_test(NAME android-video COMMAND smb3-android-video-verify)
if(SMB3_PLATFORM_TESTS_ONLY)
    return()
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(RUNNER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/tools/nesrecomp-src/runner")
if(WIN32)
    set(SDL2_DIR "${RUNNER_ROOT}/external/SDL2")
endif()
set(SMB3_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated" CACHE PATH "Private generated game code")
include("${RUNNER_ROOT}/cyc/cyc.cmake")
file(GLOB CYCLE_BANKS CONFIGURE_DEPENDS "${SMB3_GENERATED_DIR}/smb3_cyc*.c")
if(NOT EXISTS "${SMB3_GENERATED_DIR}/smb3_cyc.c" OR NOT CYCLE_BANKS)
    message(FATAL_ERROR "Missing real recompiled game code. Supply the compatible private ROM using: python tools/generate_android.py --rom <ROM.nes>. No placeholder game will be built.")
endif()
add_library(smb3-core STATIC ${NESRECOMP_CYC_SOURCES} ${CYCLE_BANKS} runtime/glue.c runtime/rom_validation.c src/widescreen.c src/wide_enemies.c)
target_include_directories(smb3-core PUBLIC ${NESRECOMP_CYC_INCLUDE_DIRS} runtime "${SMB3_GENERATED_DIR}" src)
target_link_libraries(smb3-core PUBLIC ${NESRECOMP_CYC_LIBRARIES})
set_target_properties(smb3-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    add_subdirectory(android/native)
    return()
endif()
if(WIN32)
add_executable(Smb3Recomp WIN32 src/app.c src/theme.c src/preferences.c src/choice.c src/input.c src/app.rc)
set_target_properties(Smb3Recomp PROPERTIES OUTPUT_NAME "MushroomRecompiled-v1.5.0")
target_include_directories(Smb3Recomp PRIVATE "${SDL2_DIR}/include")
target_compile_definitions(Smb3Recomp PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_directories(Smb3Recomp PRIVATE "${SDL2_DIR}/lib/x64")
target_link_libraries(Smb3Recomp PRIVATE smb3-core SDL2 user32 gdi32 comdlg32 shell32 winmm ole32 windowscodecs uuid comctl32 uxtheme msimg32 dbghelp ws2_32)
if(MINGW OR CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_link_options(Smb3Recomp PRIVATE -municode)
endif()
add_custom_command(TARGET Smb3Recomp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:Smb3Recomp>/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/gamecontrollerdb.txt" "$<TARGET_FILE_DIR:Smb3Recomp>/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/png.png" "$<TARGET_FILE_DIR:Smb3Recomp>/png.png"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDL2_DIR}/lib/x64/SDL2.dll" "$<TARGET_FILE_DIR:Smb3Recomp>")

add_executable(smb3-ui-verify tests/ui_verify.c src/theme.c src/preferences.c src/choice.c src/input.c)
target_include_directories(smb3-ui-verify PRIVATE "${SDL2_DIR}/include")
target_compile_definitions(smb3-ui-verify PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_directories(smb3-ui-verify PRIVATE "${SDL2_DIR}/lib/x64")
target_link_libraries(smb3-ui-verify PRIVATE smb3-core SDL2 user32 gdi32 comdlg32 shell32 winmm ole32 windowscodecs uuid comctl32 uxtheme msimg32)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(smb3-ui-verify PRIVATE -UNDEBUG)
endif()
endif()
add_executable(smb3-verify tests/verify.c)
target_link_libraries(smb3-verify PRIVATE smb3-core)
add_executable(smb3-wide-verify tests/widescreen_verify.c)
target_link_libraries(smb3-wide-verify PRIVATE smb3-core)
add_executable(smb3-scenes-verify tests/scenes_verify.c)
target_link_libraries(smb3-scenes-verify PRIVATE smb3-core)
add_executable(smb3-lifecycle-verify tests/lifecycle_verify.c)
target_link_libraries(smb3-lifecycle-verify PRIVATE smb3-core)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(smb3-lifecycle-verify PRIVATE -UNDEBUG)
endif()

set(SMB3_TEST_ROM "" CACHE FILEPATH "Private ROM for real gameplay regression tests")
if(SMB3_TEST_ROM)
    add_test(NAME runtime-native COMMAND smb3-verify "${SMB3_TEST_ROM}")
    add_test(NAME widescreen COMMAND smb3-wide-verify "${SMB3_TEST_ROM}")
    add_test(NAME scenes-4 COMMAND smb3-scenes-verify "${SMB3_TEST_ROM}" 4)
    add_test(NAME scenes-6 COMMAND smb3-scenes-verify "${SMB3_TEST_ROM}" 6)
    add_test(NAME lifecycle COMMAND smb3-lifecycle-verify "${SMB3_TEST_ROM}")
    set_tests_properties(runtime-native widescreen scenes-4 scenes-6 PROPERTIES RUN_SERIAL TRUE)
endif()
