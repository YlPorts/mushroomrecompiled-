set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(RUNNER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/tools/nesrecomp-src/runner")
if(WIN32)
    set(SDL2_DIR "${RUNNER_ROOT}/external/SDL2")
endif()
set(SMB3_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated" CACHE PATH "Private generated game code")
include("${RUNNER_ROOT}/cyc/cyc.cmake")
file(GLOB CYCLE_BANKS CONFIGURE_DEPENDS "${SMB3_GENERATED_DIR}/smb3_cyc*.c")
if(NOT EXISTS "${SMB3_GENERATED_DIR}/smb3_cyc.c" OR NOT CYCLE_BANKS)
    message(FATAL_ERROR "Missing real recompiled game code. Supply the compatible private ROM using: python tools/generate_android.py --rom <ROM.nes>. No placeholder game will be built.")
endif()
add_library(smb3-core STATIC ${NESRECOMP_CYC_SOURCES} ${CYCLE_BANKS} runtime/glue.c runtime/rom_validation.c src/widescreen.c src/wide_enemies.c)
target_include_directories(smb3-core PUBLIC ${NESRECOMP_CYC_INCLUDE_DIRS} runtime "${SMB3_GENERATED_DIR}" src)
target_link_libraries(smb3-core PUBLIC ${NESRECOMP_CYC_LIBRARIES})
set_target_properties(smb3-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(SMB3_OPTIMIZE_NATIVE "Optimize the cycle runtime and generated C together" ${ANDROID})
option(SMB3_LLE_RENDER_SPANS "Use equivalent stable spans in the original cycle PPU" ${SMB3_OPTIMIZE_NATIVE})
if(SMB3_LLE_RENDER_SPANS)
    target_compile_definitions(smb3-core PRIVATE NESRECOMP_LLE_RENDER_SPANS=1)
endif()
if(SMB3_OPTIMIZE_NATIVE)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT SMB3_IPO_SUPPORTED OUTPUT SMB3_IPO_ERROR LANGUAGES C)
    if(NOT SMB3_IPO_SUPPORTED)
        message(FATAL_ERROR "Native optimization requires LTO: ${SMB3_IPO_ERROR}")
    endif()
    set_target_properties(smb3-core PROPERTIES
        INTERPROCEDURAL_OPTIMIZATION_RELEASE ON
        INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO ON
        C_VISIBILITY_PRESET hidden)
    target_compile_options(smb3-core PRIVATE $<$<NOT:$<CONFIG:Debug>>:-O3>)
endif()
if(ANDROID)
    add_subdirectory(android/native)
    return()
endif()

target_include_directories(Smb3Recomp PRIVATE "${SDL2_DIR}/include")

target_link_directories(Smb3Recomp PRIVATE "${SDL2_DIR}/lib/x64")
target_link_libraries(Smb3Recomp PRIVATE smb3-core SDL2 user32 gdi32 comdlg32 shell32 winmm ole32 windowscodecs uuid comctl32 uxtheme msimg32 dbghelp ws2_32)
if(MINGW OR CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_link_options(Smb3Recomp PRIVATE -municode)
endif()

    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDL2_DIR}/lib/x64/SDL2.dll" "$<TARGET_FILE_DIR:Smb3Recomp>")
