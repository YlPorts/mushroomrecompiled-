set(SDL_SOURCE "${CMAKE_SOURCE_DIR}/build/dependencies/SDL2-2.30.12")
if(NOT EXISTS "${SDL_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run python3 tools/prepare_android.py before configuring Android")
endif()
enable_language(CXX)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDL_SOURCE}" "${CMAKE_BINARY_DIR}/sdl2")
add_library(main SHARED main.c video_layout.c "${CMAKE_SOURCE_DIR}/src/input.c")
target_include_directories(main PRIVATE "${CMAKE_SOURCE_DIR}/src")
target_link_libraries(main PRIVATE smb3-core SDL2::SDL2 android log)
target_compile_options(main PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
target_link_options(main PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
